# Qt 6.8 qsb depfiles do not escape this workspace's spaces reliably in MSBuild.
# Bake with the existing Qt shader tool and explicit dependencies (no includes).
set(_lab_shader_files)
foreach(_shader surface.frag lens.frag desktop_blur.frag desktop_edge.frag)
  set(_output "${CMAKE_CURRENT_BINARY_DIR}/optics/${_shader}.qsb")
  add_custom_command(OUTPUT "${_output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/optics"
    COMMAND Qt6::qsb --glsl "100es,120,150" --hlsl 50 --msl 12 -o "${_output}" "${CMAKE_CURRENT_SOURCE_DIR}/${_shader}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${_shader}" Qt6::qsb VERBATIM)
  set_source_files_properties("${_output}" PROPERTIES GENERATED TRUE QT_RESOURCE_ALIAS "${_shader}.qsb")
  list(APPEND _lab_shader_files "${_output}")
endforeach()
